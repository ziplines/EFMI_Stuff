void main(
  float4 v0 : SV_Position0,
  float2 v1 : TEXCOORD0,
  float2 w1 : TEXCOORD1,
  float4 v2 : TEXCOORD2,
  float4 v3 : TEXCOORD3,
  float4 v4 : TEXCOORD4,
  float4 v5 : TEXCOORD5,
  float4 v6 : TEXCOORD6,
  float4 v7 : TEXCOORD7,
  nointerpolation uint v8 : TEXCOORD8,
  uint v9 : SV_IsFrontFace0,
  out float4 o0 : SV_Target0,
  out float4 o1 : SV_Target1,
  out float4 o2 : SV_Target2,
  out float4 o3 : SV_Target3,
  out float4 o4 : SV_Target4)
{
  // ---- Editable params ----
  float Opacity = 1;                   // Base face opacity [0..1]
  float3 ColorX = float3(0.1, 0.1, 0.5); // X-dominant faces (East/West walls) [RGB 0..1]
  float3 ColorZ = float3(0.5, 0.1, 0.1); // Z-dominant faces (North/South walls) [RGB 0..1]
  float3 ColorY = float3(0.1, 0.5, 0.1); // Y-dominant faces (Top/Bottom caps) [RGB 0..1]
  // --------------------------

  float2 screenUV = v0.xy * cb1[0].zw;
  float2 ndc = screenUV * float2(2.0, -2.0) + float2(-1.0, 1.0);

  float4 clipPos;
  clipPos.xy = ndc;
  clipPos.z = v0.z;
  clipPos.w = 1.0;

  float4 worldPos = cb0[24] * clipPos.x + cb0[25] * clipPos.y + cb0[26] * clipPos.z + cb0[27];
  worldPos.xyz /= worldPos.w;

  float3 dWorldX = ddx(worldPos.xyz);
  float3 dWorldY = ddy(worldPos.xyz);
  float3 worldNormal = normalize(cross(dWorldX, dWorldY));

  float3 absNormal = abs(worldNormal);
  float3 finalColor;

  if (absNormal.x >= absNormal.y && absNormal.x >= absNormal.z)
  {
    finalColor = ColorX;
  }
  else if (absNormal.z >= absNormal.x && absNormal.z >= absNormal.y)
  {
    finalColor = ColorZ;
  }
  else
  {
    finalColor = ColorY;
  }

  o0 = float4(finalColor, Opacity);
  o1 = o2 = o3 = o4 = float4(1, 1, 1, 1);
}
